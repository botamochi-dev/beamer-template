\section{1. イントロダクション}

\begin{frame}{完全性の直感的な意味}
    本節の主目的は、\textbf{「真理（truth）」} と \textbf{「導出可能性（derivability）」} が一致することを示すことです。

    \begin{itemize}
        \item \textbf{意味論（Semantics）}: 真理値割り当てによる評価の世界。式が「正しい（妥当である）」ことを $\models \phi$ と書く。
        \item \textbf{構文論（Syntax）}: 自然演繹の世界。式が「証明可能（導出可能）である」ことを $\vdash \phi$ と書く。
    \end{itemize}
    \vspace{1em}
    2つの関係 $\models$ と $\vdash$ が完全に一致することを示します。
\end{frame}

\section{2. 健全性 (Soundness)}

\begin{frame}{補題 1.5.1：健全性定理（主張）}
    \begin{block}{補題 1.5.1（Soundness）}
        $\Gamma \vdash \phi \implies \Gamma \models \phi$
    \end{block}
    \vspace{0.5em}
    ここでは「証明図 $D$ の構成ステップに関する数学的帰納法」を用いて、すべての推論規則について真理保存性を示します。
\end{frame}

\begin{frame}{証明（詳細） — 基底段階}
    \textbf{基底段階：} 証明図 $D$ が単一の式 $\phi$ のみで構成されている場合。

    \textbf{主張：} このとき $\Gamma \vdash \phi$ ならば $\Gamma \models \phi$ が成り立つ。

    \textbf{証明：} $\phi\in\Gamma$ であり，任意の評価 $v$ が $\Gamma$ のすべての式を真にするならば， $\phi$ も真になります。よって $\Gamma\models\phi$．
\end{frame}

\begin{frame}{証明（詳細） — 帰納段階：連言 $\land$ の規則}
    \textbf{($\land I$) かつ導入:}
    
    $D$ に対して $\Gamma \models \phi$，$D'$ に対して $\Gamma' \models \phi'$ が成り立つとする。

    両方の仮定を含む集合 $\Gamma'':=\Gamma\cup\Gamma'$ を考えると，任意の評価 $v$ が $\Gamma''$ を真にする時，$v$ は $\Gamma$ と $\Gamma'$ も真にします。したがって $\phi$ と $\phi'$ は両方真になり，真理値の定義から $\phi\land\phi'$ も真になります。よって $\Gamma''\models\phi\land\phi'$．

    \textbf{($\land E$) かつ除去:}
    
    $\Gamma\models\phi\land\psi$ が成り立つとする。任意の評価 $v$ が $\Gamma$ を真にするとき $\phi\land\psi$ は真であり，ゆえに $\phi$ も真になる。すなわち $\Gamma\models\phi$（同様に $\psi$ も）。
\end{frame}

\begin{frame}{証明（詳細） — 帰納段階：含意 $\to$ の規則}
    \textbf{($\to E$) ならば除去:}
    帰納法の仮定で $\Gamma\models\phi$ と $\Gamma\models\phi\to\psi$ が成り立つとする。

    任意の評価 $v$ が $\Gamma$ を真にするなら，$\phi$ と $\phi\to\psi$ はともに真になります。真理値表より，$\phi=1$ かつ $\phi\to\psi=1$ のとき必ず $\psi=1$ でなければなりません。よって $\Gamma\models\psi$．

    \textbf{($\to I$) ならば導入:}
    
    仮定 $[\phi]^1$ を追加した状態で、証明図 $D$ によって $\psi$ を導出できたならば、「ならばの導入（$\to I$）」を適用し、仮定 $[\phi]$ をキャンセル（取り消し）して $\phi\to\psi$ を得ます。
    帰納法の仮定として $\Gamma\cup\{\phi\}\models\psi$ が成り立っているとする。

    任意の評価 $v$ が $\Gamma$ を真にする場合を考える。もし $v(\phi)=0$ ならば真理値表により $\phi\to\psi$ は自動的に 1 になる。もし $v(\phi)=1$ ならば $v$ は $\Gamma\cup\{\phi\}$ を真にするため，帰納法の仮定から $\psi$ が 1 となる。したがっていずれの場合も $\phi\to\psi$ は 1 であり，$\Gamma\models\phi\to\psi$ が示される。
\end{frame}

\begin{frame}{証明（詳細） — 帰納段階：矛盾 $\perp$ と背理法（RAA）}
    \textbf{($\perp$) 矛盾:}
    帰納法の仮定で $\Gamma\models\perp$ が成り立つとする。だが $\perp$ の評価は常に 0 であるため，
    $\Gamma$ をすべて真にする評価は存在しない。

    この場合「任意の $\Gamma$ を真にする評価 $v$ に対して $\phi$ が真になる」という命題は前提が偽であるため，任意の $\phi$ について $\Gamma\models\phi$ が成立する。

    \textbf{(RAA) 背理法:}
    帰納法の仮定として $\Gamma\cup\{\neg\phi\}\models\perp$ が成り立っているとする。

    任意の評価 $v$ が $\Gamma$ を真にする世界を取る。もし $v(\phi)=0$ だったならば $v(\neg\phi)=1$ であり，この評価は $\Gamma\cup\{\neg\phi\}$ を真にすることになる。だが帰納法の仮定によりそのような評価では $\perp$ が真になるはずだが，$\perp$ は決して真になれない。したがって仮定 $v(\phi)=0$ は矛盾であり，結論として $v(\phi)=1$ が導かれる。よって任意の $v$ に対して $\phi$ は真であり，$\Gamma\models\phi$ が成り立つ。
\end{frame}

\begin{frame}{まとめ：帰納法による健全性の完了}
    以上で示した通り、
    \begin{itemize}
        \item 最小の証明図で真理保存が成立し（基底）、
        \item すべての推論規則について「前提が意味論的に正しければ結論も正しい」ことを示した（帰納段階）、
    \end{itemize}
    よって任意の証明図 $D$ について $\Gamma\vdash\phi$ ならば $\Gamma\models\phi$ が成り立ちます。これで補題 1.5.1 の証明は完了です。\qed
\end{frame}

\begin{frame}{健全性定理の応用}
    健全性定理は地味に見えますが、\textbf{「ある命題が定理ではない（$\not\vdash \phi$）」ことを示す}のに非常に役立ちます。

    \begin{itemize}
        \item ある命題 $\phi$ が証明不可能であることを直接示すのは通常困難です（あらゆる証明図の可能性を否定する必要があるため）。
        \item しかし、健全性の対偶（$\Gamma \not\models \phi \implies \Gamma \not\vdash \phi$）を用いれば、\textbf{1つの反例（偽となる真理値割り当て）を見つけるだけでよい}ことになります。
    \end{itemize}
\end{frame}

\section{3. 無矛盾性 (Consistency)}

\begin{frame}{無矛盾性とは？}
    完全性（逆方向）の証明の礎石となるのが「無矛盾性」です。

    \begin{block}{定義 2.5.2：無矛盾性}
        命題の集合 $\Gamma$ が\textbf{「無矛盾（Consistent）」}であるとは、$\Gamma \not\vdash \bot$（$\Gamma$ から矛盾を導出できない）ことである。
    \end{block}
    
\end{frame}

\begin{frame}{補題 2.5.3：無矛盾性の同値な言い換え}
    \begin{block}{補題 2.5.3}
        以下の3つの条件は同値である：
        \begin{enumerate}
            \item[(i)] $\Gamma$ は無矛盾である。
            \item[(ii)] いかなる $\phi$ に対しても、$\Gamma \vdash \phi$ と $\Gamma \vdash \neg \phi$ の両方が成り立つことはない。
            \item[(iii)] $\Gamma \not\vdash \phi$ となる $\phi$ が少なくとも1つ存在する。
        \end{enumerate}
    \end{block}

\end{frame}

\begin{frame}{補題 2.5.4：モデルと無矛盾性の関係}
    無矛盾性を示すために「モデル」を構成する手法がよくとられます。

    \begin{block}{補題 2.5.4}
        すべての $\psi \in \Gamma$ に対して $[[\psi]]_v = 1$ となるような付値（モデル） $v$ が存在すれば、$\Gamma$ は無矛盾である。
    \end{block}
    
    \textbf{証明：}
    もし $\Gamma \vdash \bot$ なら、健全性により $\Gamma \models \bot$ となる。すると $\Gamma$ を満たす付値 $v$ は $[[\bot]]_v = 1$ を満たすはずだが、これは不可能である。よって $\Gamma$ は無矛盾。\qed
\end{frame}

\begin{frame}[fragile]{補題 2.5.5：否定と無矛盾性}
    ある集合に命題を付け加えて矛盾した場合、その命題の否定が証明されるという強力な性質です。

    \begin{block}{補題 2.5.5}
        \begin{enumerate}
            \item[(a)] $\Gamma \cup \{\neg \phi\}$ が矛盾している $\implies \Gamma \vdash \phi$
            \item[(b)] $\Gamma \cup \{\phi\}$ が矛盾している $\implies \Gamma \vdash \neg \phi$
        \end{enumerate}
    \end{block}
    
    \textbf{証明(a)：}自然演繹の背理法 (RAA) そのものです。
    \vspace{-1em}
    \begin{center}
    \begin{prooftree}
        \AxiomC{$[\neg \phi]$}
        \noLine
        \UnaryInfC{$\vdots$}
        \noLine
        \UnaryInfC{$\bot$}
        \RightLabel{(RAA)}
        \UnaryInfC{$\phi$}
    \end{prooftree}
    \end{center}
\end{frame}


\section{4. 極大無矛盾集合 (Maximally Consistent)}

\begin{frame}{定義 2.5.6：極大無矛盾集合}
    \begin{block}{定義 2.5.6}
        集合 $\Gamma$ が\textbf{「極大無矛盾（Maximally Consistent）」}であるとは：
        \begin{enumerate}
            \item[(a)] $\Gamma$ は無矛盾である。
            \item[(b)] $\Gamma \subseteq \Gamma'$ かつ $\Gamma'$ が無矛盾ならば、$\Gamma = \Gamma'$ となる。
        \end{enumerate}
    \end{block}
    
    \textbf{意味：}
    $\Gamma$ にこれ以上1つも命題を追加できない状態です。何か1つでも追加すると、途端に矛盾してしまいます。
\end{frame}

\begin{frame}{補題 2.5.7：リンデンバウムの補題）}
    任意の無矛盾な集合は、必ず極大無矛盾集合に拡張できます。

    \begin{block}{補題 2.5.7（リンデンバウムの補題）}
        任意の無矛盾な集合 $\Gamma$ は、ある極大無矛盾集合 $\Gamma^*$ に含まれる。
    \end{block}
    
    \textbf{証明：}
    あまり分からなかったので抽象的な説明だけ。
    \begin{enumerate}
        \item すべての命題を $\phi_0, \phi_1, \dots$ と一列に並べる。
        \item 順番に $\phi_i$ を見て、「追加しても無矛盾」なら追加し、「追加すると矛盾する」なら捨てる。
        \item これを無限回繰り返して得られる極限が $\Gamma^*$ となる。
    \end{enumerate}
\end{frame}

\begin{frame}{補題 2.5.8, 2.5.9：極大無矛盾集合の性質 (1)}
    $\Gamma$ を極大無矛盾集合とします。極大であるがゆえに、完璧に閉じた世界を形成します。

    \begin{block}{補題 2.5.8}
        $\Gamma$ は導出に関して閉じている。すなわち：\\
        $\Gamma \vdash \phi \implies \phi \in \Gamma$
    \end{block}

    \begin{block}{補題 2.5.9 (a)}
        任意の命題 $\phi$ について、\\
        $\phi \in \Gamma$ または $\neg \phi \in \Gamma$ の\textbf{どちらか一方のみ}が成り立つ。
    \end{block}
\end{frame}

\begin{frame}{補題 2.5.9(b), 系 2.5.10：極大無矛盾集合の性質 (2)}
    \begin{block}{補題 2.5.9 (b)}
        任意の $\phi, \psi$ について：\\
        $\phi \to \psi \in \Gamma \iff (\phi \in \Gamma \implies \psi \in \Gamma)$
    \end{block}

    \begin{block}{系 2.5.10}
        極大無矛盾集合 $\Gamma$ について、\\
        $\phi \in \Gamma \iff \neg \phi \notin \Gamma$ \quad および \quad $\neg \phi \in \Gamma \iff \phi \notin \Gamma$
    \end{block}
\end{frame}


\section{5. モデル存在補題と完全性定理}

\begin{frame}{補題 2.5.11：モデル存在補題}
    \begin{block}{補題 2.5.11}
        $\Gamma$ が無矛盾であれば、すべての $\psi \in \Gamma$ に対して $[[\psi]]_v = 1$ となるような付値（モデル） $v$ が存在する。
    \end{block}
    
    \textbf{証明のステップ：}
    \begin{enumerate}
        \item リンデンバウムの補題を用い、$\Gamma$ を極大無矛盾集合 $\Gamma^*$ に拡張。
        \item 命題変数 $p$ に対し、次のように付値 $v$ を定義する：
        $v(p) = 1 \iff p \in \Gamma^*$
        \item 帰納法により、すべての複合的な式に関しても $\phi \in \Gamma^* \iff [[\phi]]_v = 1$ を示す。
        \item $\Gamma \subseteq \Gamma^*$ なので、$\Gamma$ の式はすべて真になる。\qed
    \end{enumerate}
\end{frame}

\begin{frame}{系 2.5.12：反例モデルの存在}
    \begin{block}{系 2.5.12}
        $\Gamma \not\vdash \phi \iff$ すべての$\psi \in \Gamma$に対して$[[\psi]]_v=1$、かつ $[[\phi]]_v=0$ となるような付値 $v$ が存在する。
    \end{block}
    
    \textbf{証明の概要 ($\implies$ 側)：}
    $\Gamma \not\vdash \phi$ ならば、$\Gamma \cup \{\neg\phi\}$ は無矛盾である（補題 2.5.5からの帰結）。したがって補題2.5.11より $\Gamma \cup \{\neg\phi\}$ を満たす付値 $v$ が存在する。この $v$ は $\Gamma$ を真にし、$\phi$ を偽にする。\qed
\end{frame}

\begin{frame}{定理 2.5.13：完全性定理 (Completeness Theorem)}
    \begin{center}
        \LARGE \textbf{$\Gamma \models \phi \iff \Gamma \vdash \phi$}
    \end{center}
    
    \vspace{1em}
    \textbf{証明：}
    \begin{itemize}
        \item ($\impliedby$) は健全性（補題 2.5.1）。
        \item ($\implies$) は系 2.5.12 の対偶。\qed
    \end{itemize}
\end{frame}

\begin{frame}{完全性定理が意味するもの}
    \textbf{知的な推論と、機械的な計算の一致}
    
    \begin{itemize}
        \item 自然演繹のような「導出（$\vdash$）」を行うには適度に賢くなければならない。
        \item 一方、真理値表を用いた「妥当性の確認（$\models$）」は地道ですが、機械的・自動的に行えます。
    \end{itemize}
    \vspace{1em}
    完全性定理は、この2つが完全に同等であることを保証します。理論的には、真理値表をチェックするだけで定理を機械的に発見できるということです。
\end{frame}

\begin{frame}{決定可能性}
    完全性定理により、以下のような性質が直接的に得られます。

    \begin{block}{決定可能性 (Decidability)}
        命題論理において、任意の式 $\phi$ が証明可能（$\vdash \phi$）かどうかを判定する「アルゴリズム（有限の手順）」が存在する。
    \end{block}
    
    \textbf{理由：}
    $\vdash \phi$ かどうかを知りたければ、$\models \phi$ かどうかをチェックすれば良い。そして $\models \phi$ は、$\phi$ の完全な真理値表を書いて、最後の列がすべて $1$ かどうかを確認するアルゴリズムで確実に判定できるため。
\end{frame}

\begin{frame}{まとめ}
    本節では以下を学びました：
    
    \begin{enumerate}
        \item \textbf{健全性}：導出できることは、常に真である。
        \item \textbf{無矛盾性と極大無矛盾}：矛盾がない集合は、完璧な世界（極大無矛盾集合）へと拡張できる。
        \item \textbf{モデル存在補題}：無矛盾な集合には、それを満たす真理値割り当て（モデル）が必ず存在する。
        \item \textbf{完全性}：真である（妥当な）推論は、必ず自然演繹で導出できる。
    \end{enumerate}
\end{frame}
